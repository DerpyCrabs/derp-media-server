cmake_minimum_required(VERSION 3.24)
project(derp-media-client LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets WebEngineWidgets WebChannel Network DBus Concurrent)
find_package(KF6WindowSystem REQUIRED)
qt_add_executable(derp-media-client main.cpp client.cpp server.cpp portal.cpp resources.qrc)
target_link_libraries(derp-media-client PRIVATE Qt6::Widgets Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Network Qt6::DBus Qt6::Concurrent KF6::WindowSystem)
target_compile_options(derp-media-client PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS derp-media-client RUNTIME DESTINATION bin)
option(DERP_DESKTOP_TESTS "Build desktop integration fixtures" OFF)
if(DERP_DESKTOP_TESTS)
  qt_add_executable(derp-portal-test-parent ../../tests/desktop/parent.cpp)
  target_link_libraries(derp-portal-test-parent PRIVATE Qt6::Widgets KF6::WindowSystem)
endif()
